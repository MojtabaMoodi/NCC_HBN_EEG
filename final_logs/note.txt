  --model labram_base_patch200_200 \

  --dataset age_regression_baseline \
  --data_path "$DATA" \
  --segment_length 1s \
  --epochs 50 \
  --batch_size 1024 \
  --lr 0.0005 \
  --weight_decay 0.05 \
  --warmup_epochs 5 \
  --min_lr 1e-5 \
  --drop_path 0.1 \
  --smoothing 0.0 \
  --layer_decay 0.65 \
  --layer_scale_init_value 0.1 \
  --clip_grad 3.0 \
  --opt_betas 0.9 0.98 \

  --qkv_bias \

  --model_ema_decay 0.996 \
  --save_ckpt \
  --auto_resume \
  --seed 42 \
  --pin_mem \
  --aggregate_by_participant majority_vote \
